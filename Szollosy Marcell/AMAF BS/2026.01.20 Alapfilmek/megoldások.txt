    f.cim,
    a.nev AS rendezo,
    f.ev,
    CONCAT('https://videotorium.hu/hu/recordings/', f.filmazon) AS webcim
FROM filmek f
JOIN filmstab fs ON f.filmazon = fs.fazon
JOIN alkotok a ON fs.alkazon = a.alkotoazon
WHERE fs.munkakor = 1;

FROM filmek f
JOIN filmstab fs ON f.filmazon = fs.fazon
JOIN alkotok a ON fs.alkazon = a.alkotoazon
