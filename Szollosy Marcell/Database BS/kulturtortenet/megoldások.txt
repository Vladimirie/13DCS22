SELECT csapat.nev, SUM(megoldas.pontszam) AS osszpontszam
FROM csapat
JOIN megoldas ON csapat.id = megoldas.csapatid
GROUP BY csapat.id, csapat.nev
ORDER BY osszpontszam DESC;

5. Feladat
SELECT feladatsor.nevado,
       feladatsor.ag,
       SUM(feladat.pontszam) AS pontszam
FROM feladatsor
JOIN feladat ON feladatsor.id = feladat.feladatsorid
GROUP BY feladatsor.id, feladatsor.nevado, feladatsor.ag
HAVING SUM(feladat.pontszam) <> 150;

7. Feladat
SELECT DISTINCT csapat.nev
FROM csapat
JOIN megoldas ON csapat.id = megoldas.csapatid
JOIN feladat ON megoldas.feladatid = feladat.id
WHERE megoldas.pontszam = feladat.pontszam;

8. Feladat
SELECT feladatsor.nevado,
       COUNT(feladat.id) - COUNT(megoldas.id) AS be_nem_adott
FROM csapat
JOIN feladatsor
JOIN feladat ON feladatsor.id = feladat.feladatsorid
LEFT JOIN megoldas
  ON feladat.id = megoldas.feladatid
 AND megoldas.csapatid = csapat.id
WHERE csapat.nev = '#win'
GROUP BY feladatsor.id, feladatsor.nevado;

  AND MONTH(kituzes) = MONTH(hatarido);
