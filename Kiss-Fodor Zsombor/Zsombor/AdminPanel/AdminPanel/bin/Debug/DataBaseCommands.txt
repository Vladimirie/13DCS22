DROP TABLE IF EXISTS ingredients;
CREATE TABLE ingredients (ingredientId int AUTO_INCREMENT, ingredientName varchar(50), inStock boolean, PRIMARY KEY (ingredientID));

DROP TABLE IF EXISTS items;
CREATE TABLE items (itemID int AUTO_INCREMENT, itemName varchar(50), itemPrice int, itemIngredients varchar(50), itemAllegrenArray varchar(50), PRIMARY KEY (itemID));

DROP TABLE IF EXISTS comboItems;
CREATE TABLE comboItems (comboItemID int AUTO_INCREMENT, comboItemNane varchar(50), comboItemPrice int, itemIngredients varchar(50), PRIMARY KEY (comboItemID));

DROP TABLE IF EXISTS workers;
CREATE TABLE workers (workerID int AUTO_INCREMENT, workerFirstName varchar(50), workerLastName varchar(50), PRIMARY KEY (workerID));

DROP TABLE IF EXISTS ongoingOrders;
CREATE TABLE ongoingOrders (orderID int AUTO_INCREMENT, PRIMARY KEY (orderID));

DROP TABLE IF EXISTS completedOrders;
CREATE TABLE completedOrders (compOrderID int AUTO_INCREMENT, PRIMARY KEY (compOrderID));
